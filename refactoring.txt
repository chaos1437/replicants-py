ЦЕЛИ:
  • Чистый, легко расширяемый код (SOLID, KISS)
  • Подготовка к мультипроцессорной обработке ботов
  • Архитектура микросервиса: симуляция отдельно, UI отдельно
  • Возможность отдавать состояние мира через API (WebSocket, REST, etc.)

АРХИТЕКТУРНАЯ КОНЦЕПЦИЯ:
  Симуляция = headless микросервис, который:
    - Проводит тики симуляции
    - Хранит состояние мира
    - Предоставляет API для получения состояния
    - Не зависит от способа отображения
  
  Рендеры = отдельные клиенты (pygame, web, console, etc.)

Структура:
  def handle_spawn(world, interaction): ...
  def handle_death(world, interaction): ...
  def handle_move(world, interaction): ...
  def handle_replace(world, interaction): ...
  def handle_push_energy(world, interaction): ...
  def handle_get_energy(world, interaction): ...
  def handle_send_info(world, interaction): ...
  def handle_divide(world, interaction): ...

  HANDLERS = {
      -2: handle_spawn,
      -1: handle_death,
      0: handle_move,
      1: handle_replace,
      2: handle_push_energy,
      3: handle_get_energy,
      4: handle_send_info,
      # 5 - не нужен (обрабатывается в 4)
  }

  def get_handler(interaction_type):
      if interaction_type >= 6:
          return handle_divide
      return HANDLERS.get(interaction_type)

  def execute_interaction(world, interaction):
      handler = get_handler(interaction.type)
      if handler:
          handler(world, interaction)
      
      # Общая логика для всех
      interaction.bot.energy -= 1
      if interaction.bot.energy <= 0:
          world.remove_bot(interaction.bot)

class SimulationService:
    """Оркестратор симуляции (headless)"""
    
    def __init__(self, world, config):
        self.world = world
        self.config = config
        self.running = False
    
    def tick(self):
        """Один шаг симуляции"""
        # Подготовка к мультипроцессингу: разделяем на фазы
        
        # ФАЗА 1: Спавн новых ботов
        self._spawn_bots_if_needed()
        
        # ФАЗА 2: Обновление vision (READ-ONLY, можно параллелить)
        for bot in self.world.bots:
            self.world.update_vision_for_bot(bot)
        
        # ФАЗА 3: Выполнение ботов (можно параллелить!)
        # Каждый бот работает только со своим состоянием
        for bot in self.world.bots:
            bot.run()
        
        # ФАЗА 4: Сбор взаимодействий (можно параллелить)
        for bot in self.world.bots:
            self.world.queue_interaction(bot.get_interaction())
        
        # ФАЗА 5: Применение взаимодействий (последовательно!)
        self.world.process_interactions()
        self.world.remove_dead_bots()
        
        # Периодическое обновление
        if self.world.tick % 250 == 0:
            self._update_cells_energy()
        
        if self.world.tick % 1000 == 0:
            self._log_statistics()
    
    def run(self, max_ticks=None):
        """Запуск симуляции"""
        self.running = True
        ticks = 0
        while self.running:
            self.tick()
            ticks += 1
            if max_ticks and ticks >= max_ticks:
                break
    
    def stop(self):
        """Остановка симуляции"""
        self.running = False

Файл: services/state_provider.py

class StateProvider:
    """Предоставляет состояние мира для рендеров/API"""
    
    def __init__(self, world):
        self.world = world
    
    def get_world_state(self) -> dict:
        """Полное состояние мира для отрисовки"""
        return {
            'tick': self.world.tick,
            'width': self.world.width,
            'height': self.world.height,
            'bots': self._get_bots_state(),
            'cells': self._get_cells_state()
        }
    
    def get_bot_info(self, bot_id) -> dict:
        """Информация о конкретном боте"""
        bot = self._find_bot_by_id(bot_id)
        if not bot:
            return None
        
        return {
            'id': bot.id,
            'x': bot.x,
            'y': bot.y,
            'energy': bot.energy,
            'age': bot.age,
            'genome': {
                'program': bot.genome.program,
                'registers': bot.genome.registers
            }
        }
    
    def get_cell_state(self, x, y) -> dict:
        """Состояние ячейки"""
        cell = self.world.map.get_cell(x, y)
        if not cell:
            return None
        
        return {
            'x': x,
            'y': y,
            'energy': cell._energy,
            'occupied': cell.contains is not None,
            'bot_id': cell.contains.id if cell.contains else None
        }
    
    def get_statistics(self) -> dict:
        """Статистика симуляции"""
        return {
            'tick': self.world.tick,
            'total_bots': len(self.world.bots),
            'alive_bots': sum(1 for b in self.world.bots if b.alive),
            'average_age': sum(b.age for b in self.world.bots) / len(self.world.bots) if self.world.bots else 0
        }

Использование:
  • Pygame рендер: вызывает get_world_state() каждый кадр
  • WebSocket сервер: отправляет get_world_state() клиентам
  • REST API: endpoints для get_bot_info, get_statistics

from config.settings import load_config

from core.world_map import WorldMap

import pygame
from main import load_config, create_world
from renderers.pygame_renderer import PygameRenderer
from services.simulation_service import SimulationService
from services.state_provider import StateProvider

def main():
    config = load_config()
    world = create_world(config)
    
    # UI
    pygame.init()
    renderer = PygameRenderer(config)
    state_provider = StateProvider(world)
    
    # Симуляция с UI
    service = SimulationService(world, config)
    
    running = True
    paused = False
    
    while running:
        # Обработка событий UI
        events = renderer.handle_events()
        if events.quit:
            running = False
        if events.pause:
            paused = not paused
        
        # Тик симуляции
        if not paused:
            service.tick()
        
        # Отрисовка
        state = state_provider.get_world_state()
        renderer.render(state)

if __name__ == "__main__":
    main()

ЧЕКЛИСТ ПО ПРИНЦИПАМ ЧИСТОГО КОДА
═══════════════════════════════════════════════════════════════════════════════

После каждого изменения проверяй:

[SRP] Каждый модуль имеет одну ответственность
  ✓ Bot - поведение бота
  ✓ World - координация
  ✓ Handlers - обработка взаимодействий
  ✓ StateProvider - API состояния
  ✓ SimulationService - оркестрация

[OCP] Расширение через добавление, не изменение
  ✓ Новый тип взаимодействия = новая функция в HANDLERS
  ✓ Новый рендер = новый класс, реализующий интерфейс

[DIP] Зависимости на абстракции
  ✓ SimulationService не знает о pygame
  ✓ Core не зависит от infrastructure

[ISP] Узкие интерфейсы
  ✓ StateProvider - только get_* методы
  ✓ Handlers - функции с одной сигнатурой

[KISS] Простота
  ✓ Функции < 20 строк
  ✓ Нет magic numbers
  ✓ Понятные имена

[DRY] Нет дублирования
  ✓ Общая логика вынесена в утилиты
